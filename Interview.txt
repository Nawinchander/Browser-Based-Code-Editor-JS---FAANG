  };
}
Renderer
src/core/virtual-dom/render.js
export function render(vnode) {

  const el = document.createElement(vnode.type);

  for (const key in vnode.props) {
    el.setAttribute(key, vnode.props[key]);
  }

  vnode.children.forEach(child => {
    el.append(child);
  });

  return el;
}
Step 5 — Diffing Algorithm
Core FAANG Topic

Old Tree
vs
New Tree

Apply minimal DOM updates.

src/core/diff/diff.js
export function diff(oldNode, newNode) {

  if (!oldNode) {
    return newNode;
  }

  if (oldNode.type !== newNode.type) {
    return newNode;
  }

  return oldNode;
}

export function setState(newState) {

  store.state = {
    ...store.state,
    ...newState
  };

  store.listeners.forEach(l => l(store.state));
}

Step 7 — Event System
Pub/Sub Pattern
src/core/editor/event-bus.js
class EventBus {

  constructor() {
    this.events = {};
  }

  on(event, callback) {

    if (!this.events[event]) {
      this.events[event] = [];
    }

    this.events[event].push(callback);
  }

  emit(event, data) {

    if (!this.events[event]) return;

    this.events[event].forEach(cb => cb(data));
  }
}

export const bus = new EventBus();

function renderTree(nodes) {

  return nodes.map(node => {

    if (node.children) {
      return `
        <div>
          📁 ${node.name}
          ${renderTree(node.children)}
        </div>
      `;
    }

    return `<div>📄 ${node.name}</div>`;
  }).join('');
}

function fuzzyMatch(input, target) {

  let i = 0;

  for (const char of target) {
    if (char === input[i]) {
      i++;
    }
  }

  return i === input.length;
}

Build Index
src/services/search-index.js
export function buildIndex(files) {

  const map = {};

  files.forEach(file => {

    const words = file.content.split(/\s+/);

    words.forEach(word => {

      if (!map[word]) {
        map[word] = [];
      }

      map[word].push(file.id);
    });
  });

  return map;
}

6. What is a Piece Table?
Answer

Efficient text editing structure used in editors.

Instead of modifying large strings directly:

Original buffer remains immutable
Edits stored separately
References combined logically

Efficient for huge files.

7. Explain virtual scrolling.
Answer

Only render visible items.

Used in:

Editors
Feeds
Tables

Reduces DOM node count dramatically.
